import { prisma } from "@/app/lib/prisma";
import Link from "next/link";
import Image from "next/image";
import AddToCartButton from "@/app/components/AddToCartButton";
import StorefrontTemplate from "@/app/components/StorefrontTemplate";
import type { CSSProperties } from "react";

export const dynamic = "force-dynamic";

type StorefrontElement = {
  id: string;
  type: string;
  bgColor?: string;
  textColor?: string;
  fontSize?: number;
  textAlign?: CSSProperties["textAlign"];
  paddingY?: number;
  content?: string;
  subcontent?: string;
  imageUrl?: string;
  buttonUrl?: string;
};

type StorefrontSettings = {
  pages?: Record<string, { rawHtml?: string; rawCss?: string }>;
  canvasElements?: StorefrontElement[];
};

export default async function ShopPage() {
  // 1. Fetch Company along with its Settings
  const company = await prisma.company.findFirst({
    orderBy: { createdAt: "asc" },
    include: { companySettings: true }
  });

  // 2. Fetch Active Products
  const products = await prisma.product.findMany({
    where: {
      companyId: company?.id ?? "",
      type: { in: ["PRODUCT", "GOODS"] },
      isActive: true,
      isPublished: true,
    },
    orderBy: { createdAt: "desc" },
  });

  // 3. Extract the visual canvas layout from the database
  const settings = company?.companySettings?.storefront as StorefrontSettings | null;
  const savedPages = settings?.pages;
  const homePage = savedPages?.home;
  const elements = settings?.canvasElements ?? [];

  if (homePage?.rawHtml?.trim()) {
    return (
      <StorefrontTemplate
        html={homePage.rawHtml}
        css={homePage.rawCss}
        products={products.map((product) => ({
          id: product.id,
          name: product.name,
          imageUrl: product.imageUrl,
          salePrice: Number(product.salePrice),
          compareAtPrice: product.compareAtPrice ? Number(product.compareAtPrice) : null,
        }))}
      />
    );
  }

  // FALLBACK: If no design is saved in the admin panel yet, show the default layout
  if (elements.length === 0) {
    return (
      <main className="p-8 max-w-7xl mx-auto w-full flex flex-col">
        <div className="text-center mb-16 mt-8">
          <h1 className="text-4xl md:text-5xl font-light mb-4 text-slate-900 tracking-tight">
            The Collection
          </h1>
          <p className="text-slate-500 text-sm max-w-md mx-auto">
            Explore our latest aesthetic arrivals, curated exclusively for your workspace.
          </p>
        </div>
        {/* Fallback Grid rendered if no elements array exists */}
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-8 mb-24">
          {/* Default mapping omitted here for brevity, handled below dynamically */}
          <div className="col-span-full text-center py-24 text-slate-500 bg-slate-50 rounded-2xl border border-dashed border-slate-200">
            <p>Save a design in the Admin Theme Editor to populate this page!</p>
          </div>
        </div>
      </main>
    );
  }

  // DYNAMIC RENDERING: Map through the saved Admin Canvas Elements
  return (
    <main className="w-full flex flex-col pb-24">
      {elements.map((el) => {
        
        // ELEMENT: Announcement Bar
        if (el.type === "announcement") {
          return (
            <div 
              key={el.id}
              style={{ backgroundColor: el.bgColor, color: el.textColor, fontSize: `${el.fontSize}px`, textAlign: el.textAlign, paddingTop: `${el.paddingY}px`, paddingBottom: `${el.paddingY}px` }}
              className="w-full px-4 font-bold tracking-wide"
            >
              {el.content}
            </div>
          );
        }

        // ELEMENT: Hero Section
        if (el.type === "hero") {
          return (
            <div key={el.id} className="relative w-full min-h-[40vh] md:min-h-[60vh] flex flex-col justify-center overflow-hidden" style={{ backgroundColor: el.bgColor, paddingTop: `${el.paddingY}px`, paddingBottom: `${el.paddingY}px` }}>
              {el.imageUrl && (
                <Image src={el.imageUrl} alt="Hero Background" fill className="object-cover opacity-50 absolute inset-0 z-0" />
              )}
              <div className="relative z-10 max-w-4xl mx-auto px-6 w-full" style={{ textAlign: el.textAlign }}>
                <h1 style={{ color: el.textColor, fontSize: `${el.fontSize}px` }} className="font-bold tracking-tight mb-4 uppercase leading-tight">
                  {el.content}
                </h1>
                {el.subcontent && (
                  <p style={{ color: el.textColor }} className="text-sm md:text-lg opacity-90 max-w-2xl mx-auto">
                    {el.subcontent}
                  </p>
                )}
              </div>
            </div>
          );
        }

        // ELEMENT: Custom Text Block
        if (el.type === "text") {
          return (
            <div key={el.id} className="w-full max-w-7xl mx-auto px-6" style={{ backgroundColor: el.bgColor, paddingTop: `${el.paddingY}px`, paddingBottom: `${el.paddingY}px` }}>
              <h2 style={{ color: el.textColor, fontSize: `${el.fontSize}px`, textAlign: el.textAlign }} className="font-bold tracking-tight">
                {el.content}
              </h2>
            </div>
          );
        }

        // ELEMENT: Action Button
        if (el.type === "button") {
          return (
            <div key={el.id} className="w-full max-w-7xl mx-auto px-6" style={{ paddingTop: `${el.paddingY}px`, paddingBottom: `${el.paddingY}px`, textAlign: el.textAlign }}>
              <Link href={el.buttonUrl || "/shop"} style={{ backgroundColor: el.bgColor, color: el.textColor, fontSize: `${el.fontSize}px` }} className="inline-block px-8 py-4 rounded-xl font-bold shadow-xl hover:opacity-90 transition-opacity">
                {el.content}
              </Link>
            </div>
          );
        }

        // ELEMENT: Live Product Grid 
        if (el.type === "productGrid") {
          return (
            <div key={el.id} className="w-full max-w-7xl mx-auto px-6" style={{ backgroundColor: el.bgColor, paddingTop: `${el.paddingY}px`, paddingBottom: `${el.paddingY}px` }}>
              <h3 style={{ fontSize: `${el.fontSize}px`, textAlign: el.textAlign }} className="font-bold mb-10 text-slate-900 uppercase tracking-widest">
                {el.content}
              </h3>
              
              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-8">
                {products.map((product) => {
                  const salePrice = Number(product.salePrice);
                  const comparePrice = product.compareAtPrice ? Number(product.compareAtPrice) : null;
                  const isOnSale = comparePrice && comparePrice > salePrice;
                  const imageSource = product.imageUrl || `https://placehold.co/600x800?text=${encodeURIComponent(product.name)}`;

                  return (
                    <div key={product.id} className="group flex flex-col relative">
                      <Link href={`/shop/product/${product.id}`} className="block overflow-hidden rounded-xl bg-slate-100 aspect-[4/5] relative mb-4">
                        {isOnSale && (
                          <span className="absolute top-3 left-3 bg-rose-500 text-white text-[10px] font-bold uppercase tracking-widest px-2.5 py-1 rounded z-10 shadow-sm">
                            Sale
                          </span>
                        )}
                        <Image 
                          src={imageSource}
                          alt={product.name}
                          fill
                          className="object-cover group-hover:scale-105 transition-transform duration-700 ease-in-out"
                        />
                      </Link>
                      
                      <div className="flex items-start justify-between gap-2">
                        <Link href={`/shop/product/${product.id}`}>
                          <h3 className="text-sm font-semibold text-slate-900 hover:text-teal-600 transition-colors">
                            {product.name}
                          </h3>
                          <div className="flex items-center gap-2 mt-1">
                            <span className="text-sm font-medium text-slate-900">
                              Rs. {salePrice.toLocaleString("en-US")}
                            </span>
                            {isOnSale && (
                              <span className="text-xs text-slate-400 line-through">
                                Rs. {comparePrice.toLocaleString("en-US")}
                              </span>
                            )}
                          </div>
                        </Link>
                        
                        <AddToCartButton product={{
                          id: product.id,
                          name: product.name,
                          salePrice: salePrice,
                          imageUrl: imageSource
                        }} />
                      </div>
                    </div>
                  );
                })}

                {products.length === 0 && (
                  <div className="col-span-full text-center py-24 text-slate-500 bg-slate-50 rounded-2xl border border-dashed border-slate-200">
                    <p>No products are currently published to the storefront.</p>
                  </div>
                )}
              </div>
            </div>
          );
        }

        return null; // Ignore unknown block types
      })}
    </main>
  );
}
