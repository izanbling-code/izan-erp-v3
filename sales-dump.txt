import { useEffect, useMemo, useState } from "react";

type Customer = {

};

type Product = {
  id: string;
  name: string;
  sku?: string | null;
  costPrice?: number | string | null;
  category?: {
    id: string;
    name: string;
  } | null;
  brand?: {
    id: string;
    name: string;
  } | null;
  unit?: {
    id: string;
    name: string;
  } | null;
};

type Warehouse = {
  id: string;
  name: string;
  code?: string | null;
};

type StockRecord = {
  id: string;
  productId: string;
  warehouseId: string;
  quantity: number | string;
  averageCost?: number | string | null;
  product?: Product;
  warehouse?: Warehouse;
};

type InvoiceLine = {
  id: string;
  productId: string;
  warehouseId: string;
  quantity: number;
  unitPrice: number;
  discount: number;
  tax: number;
};

function money(value: number) {
  return `₨ ${value.toFixed(2)}`;
}

function numberValue(value: unknown) {
  const number = Number(value ?? 0);

  return Number.isFinite(number) ? number : 0;
}

function createLine(): InvoiceLine {
  return {
    id: `${Date.now()}-${Math.random()
      .toString(36)
      .slice(2)}`,
    productId: "",
    warehouseId: "",
    quantity: 1,
    unitPrice: 0,
    discount: 0,
    tax: 0,
  };
}

  const [notes, setNotes] = useState("");

            matchingStock?.averageCost
          );

        return {
          ...line,
          productId,
          unitPrice:
            averageCost > 0
              ? averageCost
              : numberValue(
                  product?.costPrice
                ),
        };
      })
    );
  }

  function handleWarehouseChange(
    lineId: string,
    warehouseId: string
  ) {
    setLines((current) =>
      current.map((line) => {
        if (line.id !== lineId) {
          return line;
        }

        return {
          ...line,
          warehouseId,
        };
      })
    );
  }

  function getAvailableStock(
    productId: string,
    warehouseId: string
  ) {
    if (!productId || !warehouseId) {
      return 0;
    }

    const record = stock.find(
      (item) =>
        item.productId === productId &&
        item.warehouseId === warehouseId
    );

    return numberValue(record?.quantity);
  }

  function getLineTotal(line: InvoiceLine) {
    const gross =
      line.quantity * line.unitPrice;

    const afterDiscount =
      Math.max(
        0,
        gross - line.discount
      );

    return afterDiscount + line.tax;
  }

  const totals = useMemo(() => {
    let subtotal = 0;
    let discount = 0;
    let tax = 0;

    for (const line of lines) {
      subtotal +=
        line.quantity *
        line.unitPrice;

      discount += line.discount;
      tax += line.tax;
    }

    const total =
      subtotal -
      discount +
      tax;

    return {
      subtotal,
      discount,
      tax,
      total,
    };
  }, [lines]);

  function validateInvoice() {
    setCustomerError("");

    if (!customerId) {
      setCustomerError(
        "Please select a customer."
      );
      return false;
    }

    if (lines.length === 0) {
      setError(
        "Please add at least one product."
      );
      return false;
    }

    for (const line of lines) {
      if (!line.productId) {
        setError(
          "Please select a product for every invoice line."
        );
        return false;
      }

      if (!line.warehouseId) {
        setError(
          "Please select a warehouse for every invoice line."
        );
        return false;
      }

      if (line.quantity <= 0) {
        setError(
          "Quantity must be greater than zero."
        );
        return false;
      }

      const available =
        getAvailableStock(
          line.productId,
          line.warehouseId
        );

      if (line.quantity > available) {
        const product =
          products.find(
            (item) =>
              item.id === line.productId
          );

        setError(
          `${product?.name || "Product"} has only ${available} available in the selected warehouse.`
        );

        return false;
      }

      if (line.unitPrice < 0) {
        setError(
          "Unit price cannot be negative."
        );
        return false;
      }

      if (line.discount < 0) {
        setError(
          "Discount cannot be negative."
        );
        return false;
      }

      if (line.tax < 0) {
        setError(
          "Tax cannot be negative."
        );
        return false;
      }
    }

    return true;
  }

function cleanString(value: unknown): string | null {
  if (typeof value !== "string") return null;
  const valueClean = value.trim();
  return valueClean.length > 0 ? valueClean : null;
}

function amountString(value: unknown): string {
  if (typeof value === "number") {
    return Number.isFinite(value) ? String(value) : "0";
  }

  if (typeof value !== "string") return "0";

  const cleaned = value.trim();
  return cleaned || "0";
}

function isValidAmount(value: string) {
  return /^\d+(\.\d{1,2})?$/.test(value) && Number(value) >= 0;
}

function errorMessage(error: unknown) {
  return error instanceof Error ? error.message : "Unexpected error.";
}

async function getCompany() {
  return prisma.company.findFirst({
    orderBy: {
      createdAt: "asc",
    },
  });
}

async function nextEntryNumber(companyId: string): Promise<string> {
  const latest = await prisma.journalEntry.findFirst({
    where: {
      companyId,
    },
    orderBy: {
      createdAt: "desc",
    },
    select: {
      entryNumber: true,
    },
  });

  const match = latest?.entryNumber?.match(/(\d+)$/);

  if (!match) return "JE-000001";

  return `JE-${String(Number(match[1]) + 1).padStart(6, "0")}`;
}

export async function GET(request: NextRequest) {
  try {
    const company = await getCompany();

    if (!company) {
      return NextResponse.json(
        {
          ok: false,
          error: "No company has been configured yet.",
        },
        { status: 400 }
      );
    }

    const { searchParams } = new URL(request.url);

    const search = cleanString(searchParams.get("search"));
    const status = cleanString(searchParams.get("status"));
    const accountId = cleanString(searchParams.get("accountId"));

    const entries = await prisma.journalEntry.findMany({
      where: {
        companyId: company.id,

        ...(status && ["DRAFT", "POSTED", "VOID"].includes(status)
          ? {
              status: status as "DRAFT" | "POSTED" | "VOID",
            }
          : {}),

        ...(accountId
          ? {
              lines: {
                some: {
                  accountId,
                },
              },
            }
          : {}),

        ...(search
          ? {
              OR: [
                {
                  entryNumber: {
                    contains: search,
                    mode: "insensitive",
                  },
                },
                {
                  reference: {
                    contains: search,
                    mode: "insensitive",
                  },
                },
                {
                  description: {
                    contains: search,
                    mode: "insensitive",
                  },
                },
              ],
            }
          : {}),
      },

      orderBy: {
        entryDate: "desc",
      },

      include: {
        lines: {
          include: {
            account: true,
          },
          orderBy: {
            id: "asc",
          },
        },
      },
    });

    const accounts = await prisma.account.findMany({
      where: {
        companyId: company.id,
        isActive: true,
      },
      orderBy: [
        {
          code: "asc",
        },
        {
          name: "asc",
        },
      ],
    });

    return NextResponse.json({

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();

    const company = await getCompany();

    if (!company) {
      return NextResponse.json(
        {
          ok: false,
          error: "No company has been configured yet.",
        },
        { status: 400 }
      );
    }

    const entryDateValue = cleanString(body.entryDate);

    if (!entryDateValue) {
      return NextResponse.json(
        {
          ok: false,
          error: "Entry date is required.",
        },
        { status: 400 }
      );
    }

    const entryDate = new Date(entryDateValue);

    if (Number.isNaN(entryDate.getTime())) {
      return NextResponse.json(
        {
          ok: false,
          error: "Invalid entry date.",
        },
        { status: 400 }
      );
    }

    const rawRows: unknown[] = Array.isArray(body.lines)
      ? body.lines
      : [];

    if (rawRows.length < 2) {
      return NextResponse.json(
        {
          ok: false,
          error: "A journal entry requires at least two lines.",
        },
        { status: 400 }
      );
    }

    const lines: JournalRow[] = rawRows.map((raw) => {
      const row =
        typeof raw === "object" && raw !== null
          ? (raw as Record<string, unknown>)
          : {};

      return {
        accountId: cleanString(row.accountId) ?? "",
        description: cleanString(row.description),
        debit: amountString(row.debit),
        credit: amountString(row.credit),
      };
    });

    for (const line of lines) {
      if (!line.accountId) {
        return NextResponse.json(
          {
            ok: false,
            error: "Every journal line must have an account.",
          },
          { status: 400 }
        );
      }

      if (!isValidAmount(line.debit) || !isValidAmount(line.credit)) {
        return NextResponse.json(
          {
            ok: false,
            error: "Debit and credit amounts must be valid numbers.",
          },
          { status: 400 }
        );
      }

      if (Number(line.debit) > 0 && Number(line.credit) > 0) {
        return NextResponse.json(
          {
            ok: false,
            error: "A journal line cannot contain both debit and credit.",
          },
          { status: 400 }
        );
      }

      if (Number(line.debit) === 0 && Number(line.credit) === 0) {
        return NextResponse.json(
          {
            ok: false,
            error: "Every journal line must contain a debit or credit amount.",
          },
          { status: 400 }
        );
      }
    }

    const accountIds = [
      ...new Set(lines.map((line) => line.accountId)),
    ];

    const accounts = await prisma.account.findMany({
      where: {
        companyId: company.id,
        id: {
          in: accountIds,
        },
        isActive: true,
      },
    });

    if (accounts.length !== accountIds.length) {
      return NextResponse.json(
        {
          ok: false,
          error: "One or more selected accounts are invalid.",
        },
        { status: 400 }
      );
    }

    const totalDebit = lines.reduce(
      (sum, line) => sum + Number(line.debit),
      0
    );

    const totalCredit = lines.reduce(
      (sum, line) => sum + Number(line.credit),
      0
    );

    const requestedStatus =
      body.status === "POSTED" ? "POSTED" : "DRAFT";

    if (
      requestedStatus === "POSTED" &&
      Math.abs(totalDebit - totalCredit) > 0.005
    ) {
      return NextResponse.json(
        {
          ok: false,
          error: `Journal entry is not balanced. Debit: ${totalDebit.toFixed(
            2
          )}, Credit: ${totalCredit.toFixed(2)}.`,
        },
        { status: 400 }
      );
    }

    const entryNumber = await nextEntryNumber(company.id);

    const journalEntry = await prisma.journalEntry.create({
      data: {
        companyId: company.id,
        entryNumber,
        entryDate,
        reference: cleanString(body.reference),
        description: cleanString(body.description),
        status: requestedStatus,
        lines: {
          create: lines.map((line) => ({
            accountId: line.accountId,
            description: line.description,
            debit: line.debit,
            credit: line.credit,
          })),
        },
      },

      include: {
        lines: {
          include: {
            account: true,
          },
        },
      },
    });

    return NextResponse.json(
      {
        ok: true,
        message:
          requestedStatus === "POSTED"
            ? "Journal entry posted successfully."
            : "Journal entry saved as draft.",
        entry: journalEntry,
      },
      { status: 201 }
    );
  } catch (error) {
    console.error("POST /api/accounting/journals error:", error);

    return NextResponse.json(
      {
        ok: false,
        error: errorMessage(error),
      },
      { status: 500 }
    );
  }
}

const MOVEMENT_TYPES = [
  "ADJUSTMENT_IN",
  "ADJUSTMENT_OUT",
  "TRANSFER_IN",

  "OPENING",

type MovementType = (typeof MOVEMENT_TYPES)[number];

function isMovementType(value: unknown): value is MovementType {
  return (
    typeof value === "string" &&
    (MOVEMENT_TYPES as readonly string[]).includes(value)
  );
}

function toNumber(value: unknown): number {
  const n = Number(value ?? 0);
  return Number.isFinite(n) ? n : 0;
}

function cleanString(value: unknown): string | null {
  if (value === null || value === undefined) {
    return null;
  }

  const valueString = String(value).trim();
  return valueString.length > 0 ? valueString : null;
}

function parseDate(value: unknown): Date {
  if (!value) {
    return new Date();
  }

  const date = new Date(String(value));

  return Number.isNaN(date.getTime()) ? new Date() : date;
}

function generateBatchNumber(prefix: string): string {
  const now = new Date();

  const datePart = now
    .toISOString()
    .replace(/[-:TZ.]/g, "")
    .slice(0, 14);

  const randomPart = Math.random()
    .toString(36)
    .substring(2, 8)
    .toUpperCase();

  return `${prefix}-${datePart}-${randomPart}`;
}

async function createOpeningStock(body: any) {
  const productId = String(body.productId ?? "");
  const warehouseId = String(
    body.warehouseId ?? ""
  );
  const batchNumber = String(
    body.batchNumber ?? ""
  ).trim();

  const quantity = decimalNumber(body.quantity);
  const unitCost = decimalNumber(body.unitCost);

  if (!productId) {
    return NextResponse.json(
      {
        ok: false,
        error: "Product is required.",
      },
      { status: 400 }
    );
  }

  if (!warehouseId) {
    return NextResponse.json(
      {
        ok: false,
        error: "Warehouse is required.",
      },
      { status: 400 }
    );
  }

  if (!batchNumber) {
    return NextResponse.json(
      {
        ok: false,
        error: "Batch number is required.",
      },
      { status: 400 }
    );
  }

  if (!Number.isFinite(quantity) || quantity <= 0) {
    return NextResponse.json(
      {
        ok: false,
        error: "Quantity must be greater than zero.",
      },
      { status: 400 }
    );
  }

  if (!Number.isFinite(unitCost) || unitCost < 0) {
    return NextResponse.json(
      {
        ok: false,
        error: "Unit cost cannot be negative.",
      },
      { status: 400 }
    );
  }

  const company = await prisma.company.findFirst({
    orderBy: {
      createdAt: "asc",
    },
  });

  if (!company) {
    return NextResponse.json(
      {
        ok: false,
        error: "No company has been configured yet",
      },
      { status: 400 }
    );
  }

  const result = await prisma.$transaction(
    async (tx) => {
      const product = await tx.product.findFirst({
        where: {
          id: productId,
          companyId: company.id,
        },
      });

      if (!product) {
        throw new Error("Product not found.");
      }

      if (product.type !== "PRODUCT") {
        throw new Error(
          "Opening stock can only be added to products, not services."
        );
      }

      if (!product.isActive) {
        throw new Error(
          "Cannot add opening stock to an inactive product."
        );
      }

      const warehouse =
        await tx.warehouse.findFirst({
          where: {
            id: warehouseId,
            companyId: company.id,
            isActive: true,
          },
        });

      if (!warehouse) {
        throw new Error(
          "Selected warehouse was not found or is inactive."
        );
      }

      const existingBatch =
        await tx.inventoryBatch.findFirst({
          where: {
            companyId: company.id,
            productId,
            batchNumber,
          },
          select: {
            id: true,
          },
        });

      if (existingBatch) {
        throw new Error(
          `Batch "${batchNumber}" already exists for this product.`
        );
      }

      const purchaseDate = body.purchaseDate
        ? new Date(body.purchaseDate)
        : new Date();

      if (Number.isNaN(purchaseDate.getTime())) {
        throw new Error(
          "Invalid opening stock date."
        );
      }

      const batch =
        await tx.inventoryBatch.create({
          data: {
            companyId: company.id,
            productId,
            batchNumber,
            purchaseDate,
            unitCost,
            originalQuantity: quantity,
            isActive: true,
          },
        });

      await tx.stockBatch.create({
        data: {
          batchId: batch.id,
          productId,
          warehouseId,
          quantity,
        },
      });

      const existingStock =
        await tx.stock.findUnique({
          where: {
            productId_warehouseId: {
              productId,
              warehouseId,
            },
          },
        });

      const oldQuantity = Number(
        existingStock?.quantity ?? 0
      );

      const oldAverageCost = Number(
        existingStock?.averageCost ?? 0
      );

      const newQuantity =
        oldQuantity + quantity;

      const newAverageCost =
        newQuantity > 0
          ? (oldQuantity * oldAverageCost +
              quantity * unitCost) /
            newQuantity
          : unitCost;

      await tx.stock.upsert({
        where: {
          productId_warehouseId: {
            productId,
            warehouseId,
          },
        },
        create: {
          productId,
          warehouseId,
          quantity,
          averageCost: unitCost,
        },
        update: {
          quantity: newQuantity,
          averageCost: newAverageCost,
        },
      });

      const totalCost =
        quantity * unitCost;

      const movement =
        await tx.inventoryMovement.create({
          data: {
            companyId: company.id,
            productId,
            batchId: batch.id,
            destinationWarehouseId: warehouseId,
            type: "OPENING",
            referenceType: "OPENING",
            referenceId: batch.id,
            quantity,
            unitCost,
            totalCost,
            movementDate: purchaseDate,
            notes:
              "Opening inventory balance",
          },
        });

      return {
        batch,
        movement,
      };
    }
  );

  return NextResponse.json(
    {
      ok: true,
      batch: result.batch,
      movement: result.movement,

          continue;
        }

        const currentQuantity = Number(stock.quantity ?? 0);
        const batchQuantity = Number(stockBatch.quantity ?? 0);

        const newQuantity =
          currentQuantity - batchQuantity;

                productId: batch.productId,
                warehouseId: stockBatch.warehouseId,

        },

        },

async function createOpeningStock(body: any) {
  const productId = String(body.productId ?? "");
  const warehouseId = String(
    body.warehouseId ?? ""
  );
  const batchNumber = String(
    body.batchNumber ?? ""
  ).trim();

  const quantity = decimalNumber(body.quantity);
  const unitCost = decimalNumber(body.unitCost);

  if (!productId) {
    return NextResponse.json(
      {
        ok: false,
        error: "Product is required.",
      },
      { status: 400 }
    );
  }

  if (!warehouseId) {
    return NextResponse.json(
      {
        ok: false,
        error: "Warehouse is required.",
      },
      { status: 400 }
    );
  }

  if (!batchNumber) {
    return NextResponse.json(
      {
        ok: false,
        error: "Batch number is required.",
      },
      { status: 400 }
    );
  }

  if (!Number.isFinite(quantity) || quantity <= 0) {
    return NextResponse.json(
      {
        ok: false,
        error: "Quantity must be greater than zero.",
      },
      { status: 400 }
    );
  }

  if (!Number.isFinite(unitCost) || unitCost < 0) {
    return NextResponse.json(
      {
        ok: false,
        error: "Unit cost cannot be negative.",
      },
      { status: 400 }
    );
  }

  const company = await prisma.company.findFirst({
    orderBy: {
      createdAt: "asc",
    },
  });

  if (!company) {
    return NextResponse.json(
      {
        ok: false,
        error: "No company has been configured yet",
      },
      { status: 400 }
    );
  }

  const result = await prisma.$transaction(
    async (tx) => {
      const product = await tx.product.findFirst({
        where: {
          id: productId,
          companyId: company.id,
        },
      });

      if (!product) {
        throw new Error("Product not found.");
      }

      if (product.type !== "PRODUCT") {
        throw new Error(
          "Opening stock can only be added to products, not services."
        );
      }

      if (!product.isActive) {
        throw new Error(
          "Cannot add opening stock to an inactive product."
        );
      }

      const warehouse =
        await tx.warehouse.findFirst({
          where: {
            id: warehouseId,
            companyId: company.id,
            isActive: true,
          },
        });

      if (!warehouse) {
        throw new Error(
          "Selected warehouse was not found or is inactive."
        );
      }

      const existingBatch =
        await tx.inventoryBatch.findFirst({
          where: {
            companyId: company.id,
            productId,
            batchNumber,
          },
          select: {
            id: true,
          },
        });

      if (existingBatch) {
        throw new Error(
          `Batch "${batchNumber}" already exists for this product.`
        );
      }

      const purchaseDate = body.purchaseDate
        ? new Date(body.purchaseDate)
        : new Date();

      if (Number.isNaN(purchaseDate.getTime())) {
        throw new Error(
          "Invalid opening stock date."
        );
      }

      const batch =
        await tx.inventoryBatch.create({
          data: {
            companyId: company.id,
            productId,
            batchNumber,
            purchaseDate,
            unitCost,
            originalQuantity: quantity,
            isActive: true,
          },
        });

      await tx.stockBatch.create({
        data: {
          batchId: batch.id,
          productId,
          warehouseId,
          quantity,
        },
      });

      const existingStock =
        await tx.stock.findUnique({
          where: {
            productId_warehouseId: {
              productId,
              warehouseId,
            },
          },
        });

      const oldQuantity = Number(
        existingStock?.quantity ?? 0
      );

      const oldAverageCost = Number(
        existingStock?.averageCost ?? 0
      );

      const newQuantity =
        oldQuantity + quantity;

      const newAverageCost =
        newQuantity > 0
          ? (oldQuantity * oldAverageCost +
              quantity * unitCost) /
            newQuantity
          : unitCost;

      await tx.stock.upsert({
        where: {
          productId_warehouseId: {
            productId,
            warehouseId,
          },
        },
        create: {
          productId,
          warehouseId,
          quantity,
          averageCost: unitCost,
        },
        update: {
          quantity: newQuantity,
          averageCost: newAverageCost,
        },
      });

      const totalCost =
        quantity * unitCost;

      const movement =
        await tx.inventoryMovement.create({
          data: {
            companyId: company.id,
            productId,
            batchId: batch.id,
            destinationWarehouseId: warehouseId,
            type: "OPENING",
            referenceType: "OPENING",
            referenceId: batch.id,
            quantity,
            unitCost,
            totalCost,
            movementDate: purchaseDate,
            notes:
              "Opening inventory balance",
          },
        });

      return {
        batch,
        movement,
      };
    }
  );

  return NextResponse.json(
    {
      ok: true,
      batch: result.batch,
      movement: result.movement,
    },
    { status: 201 }
  );

/* ============================================================

   ============================================================ */

function decimalNumber(value: unknown): number {
  const number = Number(value ?? 0);

  if (!Number.isFinite(number)) {
    return 0;
  }

  return number;
}

function cleanString(value: unknown): string | null {
  if (value === undefined || value === null) {
    return null;
  }

  const result = String(value).trim();

  return result.length > 0 ? result : null;
}

function parseDate(
  value: unknown,
  fallback: Date = new Date()
): Date {
  if (!value) {
    return fallback;
  }

  const date = new Date(String(value));

  if (Number.isNaN(date.getTime())) {
    return fallback;
  }

  return date;
}

function calculateLineTotal(line: PurchaseLineInput) {
  const quantity = decimalNumber(line.quantity);
  const unitCost = decimalNumber(line.unitCost);
  const discount = decimalNumber(line.discount);
  const tax = decimalNumber(line.tax);

  const gross = quantity * unitCost;

  const netBeforeTax = Math.max(0, gross - discount);

  return netBeforeTax + tax;
}

function generateBatchNumber() {
  const now = new Date();

  const datePart = now
    .toISOString()
    .replace(/[-:TZ.]/g, "")
    .slice(0, 14);

  const randomPart = Math.random()
    .toString(36)
    .substring(2, 7)
    .toUpperCase();

  return `B-${datePart}-${randomPart}`;
}

      return NextResponse.json(

          error: "At least one purchase item is required",
        },
        {
          status: 400,
        }
      );
    }

    /*
     * --------------------------------------------------------
     * TEMPORARY SINGLE COMPANY CONTEXT
     * --------------------------------------------------------
     */

    const company = await prisma.company.findFirst({
      orderBy: {
        createdAt: "asc",
      },
    });

    if (!company) {
      return NextResponse.json(
        {
          ok: false,
          error: "No company has been configured",
        },
        {
          status: 400,
        }
      );
    }

    if (!company) {
      throw new Error("No company has been configured yet.");
    }

    /*
     * --------------------------------------------------------
     * SUPPLIER
     * --------------------------------------------------------
     */

    const supplier = await prisma.supplier.findFirst({
      where: {
        id: supplierId,

     * --------------------------------------------------------

      const unitCost = decimalNumber(line.unitCost);
      const discount = decimalNumber(line.discount);

        throw new Error(
          "Every purchase line must have a product"
        );

      if (!line.warehouseId) {
        throw new Error(
          "Every purchase line must have a warehouse"
        );
      }

      if (quantity <= 0) {
        throw new Error(
          "Purchase quantity must be greater than zero"
        );
      }

      if (unitCost < 0) {
        throw new Error(
          "Purchase cost cannot be negative"
        );
      }

      if (discount < 0) {
        throw new Error(
          "Discount cannot be negative"
        );
      }

      if (tax < 0) {
        throw new Error(
          "Tax cannot be negative"
        );
      }

      return {
        ...line,
        quantity,
        unitCost,
        discount,
        tax,
        total:
          quantity * unitCost -
          discount +
          tax,
      };
    });

     * --------------------------------------------------------
     * VALIDATE PRODUCTS AND WAREHOUSES
     * --------------------------------------------------------
     */

    const productIds = [
      ...new Set(lines.map((line) => line.productId)),
    ];

    const warehouseIds = [
      ...new Set(lines.map((line) => line.warehouseId)),
    ];

    const products = await prisma.product.findMany({
      where: {
        companyId: company.id,
        id: {
          in: productIds,
        },
      },
    });

    const warehouses = await prisma.warehouse.findMany({
      where: {
        companyId: company.id,
        id: {
          in: warehouseIds,
        },
      },

          status: 400,
        }
      );

          status: 400,
        }
      );

     * CALCULATE TOTALS
     * --------------------------------------------------------

        sum +
        line.quantity * line.unitCost,
      0
    );

      (sum, line) =>
        sum + line.discount,
      0
    );

      (sum, line) =>
        sum + line.tax,

function decimalNumber(value: unknown): number {
  const number = Number(value ?? 0);

  if (!Number.isFinite(number)) {
    return 0;
  }

  return number;
}

function cleanString(value: unknown): string | null {
  if (value === undefined || value === null) {
    return null;
  }

  const result = String(value).trim();

  return result.length > 0 ? result : null;
}

function parseDate(
  value: unknown,
  fallback: Date = new Date()
): Date {
  if (!value) {
    return fallback;
  }

  const date = new Date(String(value));

  if (Number.isNaN(date.getTime())) {
    return fallback;
  }

  return date;
}

function calculateLineTotal(line: PurchaseLineInput) {
  const quantity = decimalNumber(line.quantity);
  const unitCost = decimalNumber(line.unitCost);
  const discount = decimalNumber(line.discount);
  const tax = decimalNumber(line.tax);

  const gross = quantity * unitCost;

  const netBeforeTax = Math.max(0, gross - discount);

  return netBeforeTax + tax;
}

function generateBatchNumber() {
  const now = new Date();

  const datePart = now
    .toISOString()
    .replace(/[-:TZ.]/g, "")
    .slice(0, 14);

  const randomPart = Math.random()
    .toString(36)
    .substring(2, 7)
    .toUpperCase();

  return `B-${datePart}-${randomPart}`;
}

                ),
                dueDate: body.dueDate
                  ? parseDate(body.dueDate)
                  : null,

                status,

                subtotal,
                discount: totalDiscount,
                tax: totalTax,
                total,

                paid: 0,
                balance: total,

                notes:
                  cleanString(body.notes),

                lines: {
                  create: [],
                },
              },
            });

          /*
           * --------------------------------------------------
           * PROCESS EACH PURCHASE LINE
           * --------------------------------------------------
           */

          for (const line of lines) {
            /*
             * Find product.
             */

            const product =
              products.find(
                (item) =>
                  item.id === line.productId
              );

            if (!product) {
              throw new Error(
                "Product not found"
              );
            }

            /*
             * ------------------------------------------------
             * BATCH
             * ------------------------------------------------
             *
             * If a batchId was supplied, use the existing batch.
             *
             * Otherwise create a new batch.
             */

            let batch;

            if (line.batchId) {
              batch =
                await tx.inventoryBatch.findFirst(
                  {
                    where: {
                      id: line.batchId,
                      companyId: company.id,
                      productId: product.id,
                    },
                  }
                );

              if (!batch) {
                throw new Error(
                  `Batch not found for product ${product.name}`
                );
              }

              /*
               * Existing batch:
               *
               * We intentionally DO NOT silently change its
               * cost because a batch represents one historical
               * cost layer.
               *
               * If a different cost is supplied, reject it.
               */

              if (
                Number(batch.unitCost) !==
                Number(line.unitCost)
              ) {
                throw new Error(
                  `Batch ${batch.batchNumber} already has a different unit cost`
                );
              }
            } else {
              /*
               * Create a new batch.
               */

              let batchNumber =
                cleanString(
                  line.batchNumber
                );

              if (!batchNumber) {
                batchNumber =
                  generateBatchNumber();
              }

              /*
               * Make sure the generated batch number
               * is unique for this company/product.
               */

              let existing =
                await tx.inventoryBatch.findFirst(
                  {
                    where: {
                      companyId: company.id,
                      productId: product.id,
                      batchNumber,
                    },
                  }
                );

              while (existing) {
                batchNumber =
                  generateBatchNumber();

                existing =
                  await tx.inventoryBatch.findFirst(
                    {
                      where: {
                        companyId: company.id,
                        productId: product.id,
                        batchNumber,
                      },
                    }
                  );
              }

              batch =
                await tx.inventoryBatch.create(
                  {
                    data: {
                      companyId:
                        company.id,

                      productId:
                        product.id,

                      batchNumber,

                      purchaseBillId:
                        bill.id,

                      supplierId:
                        supplier.id,

                      purchaseDate:
                        bill.billDate,

                      unitCost:
                        line.unitCost,

                      originalQuantity:
                        line.quantity,

                      isActive: true,
                    },
                  }
                );
            }

            /*
             * ------------------------------------------------
             * PURCHASE BILL LINE
             * ------------------------------------------------
             */

            await tx.purchaseBillLine.create(
              {
                data: {
                  billId: bill.id,

                  productId:
                    product.id,

                  batchId:
                    batch.id,

                  description:
                    cleanString(
                      line.description
                    ),

                  quantity:
                    line.quantity,

                  unitCost:
                    line.unitCost,

                  discount:
                    line.discount,

                  tax:
                    line.tax,

                  total:
                    line.total,

                  warehouseId:
                    line.warehouseId,
                },
              }
            );

            /*
             * ------------------------------------------------
             * ONLY POSTED PURCHASES AFFECT INVENTORY
             * ------------------------------------------------
             */

            if (status === "POSTED") {
              /*
               * ----------------------------------------------
               * BATCH STOCK
               * ----------------------------------------------
               */

              const existingBatchStock =
                await tx.stockBatch.findUnique(
                  {
                    where: {
                      batchId_warehouseId: {
                        batchId:
                          batch.id,

                        warehouseId:
                          line.warehouseId,
                      },
                    },
                  }
                );

              if (
                existingBatchStock
              ) {
                await tx.stockBatch.update(
                  {
                    where: {
                      id:
                        existingBatchStock.id,
                    },

                    data: {
                      quantity: {
                        increment:
                          line.quantity,
                      },
                    },
                  }
                );
              } else {
                await tx.stockBatch.create(
                  {
                    data: {
                      batchId:
                        batch.id,

                      productId:
                        product.id,

                      warehouseId:
                        line.warehouseId,

                      quantity:
                        line.quantity,
                    },
                  }
                );
              }

              /*
               * ----------------------------------------------
               * TOTAL PRODUCT STOCK
               * ----------------------------------------------
               */

              const existingStock =
                await tx.stock.findUnique(
                  {
                    where: {
                      productId_warehouseId: {
                        productId:
                          product.id,

                        warehouseId:
                          line.warehouseId,
                      },
                    },
                  }
                );

              if (existingStock) {
                const oldQuantity =
                  Number(
                    existingStock.quantity
                  );

                const oldAverageCost =
                  Number(
                    existingStock.averageCost
                  );

                const newQuantity =
                  oldQuantity +
                  line.quantity;

                const newAverageCost =
                  newQuantity > 0
                    ? (
                        oldQuantity *
                          oldAverageCost +
                        line.quantity *
                          line.unitCost
                      ) /
                      newQuantity
                    : line.unitCost;

                await tx.stock.update(
                  {
                    where: {
                      id:
                        existingStock.id,
                    },

                    data: {
                      quantity:
                        newQuantity,

                      averageCost:
                        newAverageCost,
                    },
                  }
                );
              } else {
                await tx.stock.create(
                  {
                    data: {
                      productId:
                        product.id,

                      warehouseId:
                        line.warehouseId,

                      quantity:
                        line.quantity,

                      averageCost:
                        line.unitCost,
                    },
                  }
                );
              }

              /*
               * ----------------------------------------------
               * INVENTORY MOVEMENT
               * ----------------------------------------------
               */

              await tx.inventoryMovement.create(
                {
                  data: {
                    companyId:
                      company.id,

                    productId:
                      product.id,

                    batchId:
                      batch.id,

                    destinationWarehouseId:
                      line.warehouseId,

                    type: "PURCHASE",

                    referenceType:
                      "PURCHASE",

                    referenceId:
                      bill.id,

                    quantity:
                      line.quantity,

                    unitCost:
                      line.unitCost,

                    totalCost:
                      line.quantity *
                      line.unitCost,

                    movementDate:
                      bill.billDate,

                    notes:
                      `Purchase Bill ${bill.billNo}`,
                  },
                }
              );
            }
          }

          /*
           * --------------------------------------------------
           * RETURN COMPLETE BILL
           * --------------------------------------------------
           */

          return tx.purchaseBill.findUnique(
            {
              where: {
                id: bill.id,
              },

              include: {
                supplier: true,

                lines: {
                  include: {
                    product: true,
                    batch: true,
                    warehouse: true,
                  },
                },
              },
            }
          );
        }
      );

    return NextResponse.json(
      {
        ok: true,
        bill: result,
      },
      {
        status: 201,
      }
    );
  } catch (error) {
    console.error(
      "POST /api/purchases/bills error:",
      error
    );

    const message =
      error instanceof Error
        ? error.message
        : "Failed to create purchase bill";

    return NextResponse.json(
      {
        ok: false,
        error: message,
      },
      {
        status: 500,
      }
    );
  }
}

export async function PUT(request: NextRequest) {

      (await request.json()) as PurchaseBillInput;

        {
          ok: false,
          error: "Purchase bill ID is required",
        },
        {
          status: 400,
        }
      );
    }

    const existing =
      await prisma.purchaseBill.findUnique({
        where: {
          id: body.id,
        },

        include: {
          lines: true,
        },
      });

    if (!existing) {
      return NextResponse.json(
        {
          ok: false,
          error: "Purchase bill not found",
        },
        {
          status: 404,
        }
      );
    }

    /*
     * --------------------------------------------------------
     * IMPORTANT
     * --------------------------------------------------------
     *
     * Posted bills are protected from ordinary editing.
     *
     * This prevents inventory corruption.
     *
     * Later we can implement a proper "Reverse and Edit"
     * workflow.
     */

    if (
      existing.status === "POSTED" ||
      existing.status === "PARTIAL" ||
      existing.status === "PAID"
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Posted purchase bills cannot be edited directly. Reverse the bill first.",
        },
        {
          status: 409,
        }
      );
    }

    if (!body.supplierId) {
      return NextResponse.json(
        {
          ok: false,
          error: "Supplier is required",
        },
        {
          status: 400,
        }
      );
    }

    if (
      !Array.isArray(body.lines) ||
      body.lines.length === 0
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "At least one purchase line is required",
        },
        {
          status: 400,
        }
      );
    }

    const company =
      await prisma.company.findFirst({
        orderBy: {
          createdAt: "asc",
        },
      });

    if (!company) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "No company has been configured",
        },
        {
          status: 400,
        }
      );
    }

    if (!company) {
      throw new Error("No company has been configured yet.");
    }

    const supplier =
      await prisma.supplier.findFirst({
        where: {
          id: body.supplierId,
          companyId: company.id,
        },
      });

    if (!supplier) {
      return NextResponse.json(
        {
          ok: false,
          error: "Supplier not found",
        },
        {
          status: 404,
        }
      );
    }

    const lines = body.lines.map(
      (line) => {
        const quantity =
          decimalNumber(
            line.quantity
          );

        const unitCost =
          decimalNumber(
            line.unitCost
          );

        const discount =
          decimalNumber(
            line.discount
          );

        const tax =
          decimalNumber(line.tax);

        if (quantity <= 0) {
          throw new Error(
            "Quantity must be greater than zero"
          );
        }

        if (unitCost < 0) {
          throw new Error(
            "Unit cost cannot be negative"
          );
        }

        if (!line.productId) {
          throw new Error(
            "Product is required"
          );
        }

        if (!line.warehouseId) {
          throw new Error(
            "Warehouse is required"
          );
        }

        return {
          ...line,
          quantity,
          unitCost,
          discount,
          tax,
          total:
            quantity * unitCost -
            discount +
            tax,
        };
      }
    );

    const subtotal =
      lines.reduce(
        (sum, line) =>
          sum +
          line.quantity *
            line.unitCost,

      lines.reduce(

          sum + line.discount,

      decimalNumber(body.tax);

    const discount =
      lineDiscount +
      billDiscount;

    const tax =
      lineTax +
      billTax;

    const total =
      subtotal -
      discount +
      tax;

                tax,
                total,

                  total -
                  Number(existing.paid),

                notes:
                  cleanString(
                    body.notes
                  ),
              },
            }
          );

          /*
           * Recreate lines.
           */

          for (const line of lines) {
            let batch;

            if (line.batchId) {
              batch =
                await tx.inventoryBatch.findFirst(
                  {
                    where: {
                      id:
                        line.batchId,

                      companyId:
                        company.id,

                      productId:
                        line.productId,
                    },
                  }
                );

              if (!batch) {
                throw new Error(
                  "Selected batch not found"
                );
              }

              if (
                Number(
                  batch.unitCost
                ) !==
                Number(
                  line.unitCost
                )
              ) {
                throw new Error(

                );
              }
            } else {
              let batchNumber =
                cleanString(
                  line.batchNumber
                ) ??
                generateBatchNumber();

              let existingBatch =
                await tx.inventoryBatch.findFirst(
                  {
                    where: {
                      companyId:
                        company.id,

                      productId:
                        line.productId,

                      batchNumber,
                    },
                  }
                );

              while (existingBatch) {

                  generateBatchNumber();

                    {
                      where: {
                        companyId:
                          company.id,

                        productId:
                          line.productId,

                        batchNumber,

                        line.productId,

                      batchNumber,

                      purchaseBillId: String(body.id),

export async function DELETE(
  request: NextRequest
) {
  try {
    const { searchParams } =
      new URL(request.url);

    const id =
      searchParams.get("id");

    if (!id) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Purchase bill ID is required",
        },
        {
          status: 400,
        }
      );
    }

    const bill =
      await prisma.purchaseBill.findUnique(
        {
          where: {
            id,
          },

          include: {
            allocations: true,
            lines: true,
          },
        }
      );

    if (!bill) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Purchase bill not found",
        },
        {
          status: 404,
        }
      );
    }

    /*
     * --------------------------------------------------------
     * PROTECT POSTED BILLS
     * --------------------------------------------------------
     *
     * A posted bill may already have:
     *
     * - stock
     * - batches
     * - inventory movements
     * - supplier payable
     * - journal entries
     *
     * Therefore it cannot simply be deleted.
     */

    if (
      bill.status === "POSTED" ||
      bill.status === "PARTIAL" ||
      bill.status === "PAID"
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Posted purchase bills cannot be deleted. Reverse the purchase first.",
        },
        {
          status: 409,
        }
      );
    }

    if (bill.allocations.length > 0) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "This purchase bill has payments allocated to it and cannot be deleted.",
        },
        {
          status: 409,
        }
      );
    }

    await prisma.$transaction(
      async (tx) => {
        /*
         * Draft lines cascade from the bill.
         */

        await tx.purchaseBill.delete({
          where: {
            id,
          },
        });

        /*
         * Remove batches that were created specifically
         * by this draft purchase and have never been used.
         */

        await tx.inventoryBatch.deleteMany({
          where: {
            purchaseBillId: id,

            stockBatches: {
              none: {
                quantity: {
                  gt: 0,
                },
              },
            },

            salesLines: {
              none: {},
            },
          },
        });
      }
    );

    return NextResponse.json({
      ok: true,
      message:
        "Purchase bill deleted successfully",
    });
  } catch (error) {
    console.error(
      "DELETE /api/purchases/bills error:",
      error
    );

    return NextResponse.json(
      {
        ok: false,
        error:
          error instanceof Error
            ? error.message
            : "Failed to delete purchase bill",
      },
      {
        status: 500,
      }
    );
  }
}

// ============================================================
// INVENTORY BATCH
//
// One batch represents one cost layer.
//
// Example:
//
// Product: Watch ABC
// Batch: WB-001
// Cost: PKR 1,000
// Purchased: 20
//
// Product: Watch ABC
// Batch: WB-002
// Cost: PKR 1,200
// Purchased: 15
//
// These remain separate cost layers.
// ============================================================

model InventoryBatch {
  id        String @id @default(cuid())
  companyId String
  productId String

  /**
   * Human-readable batch identifier.
   * Example:
   * WB-2026-001
   */
  batchNumber String

  purchaseBillId String?
  supplierId     String?

  purchaseDate DateTime @default(now())

  /**
   * Cost per unit for this batch.
   * This is the cost used for COGS.
   */
  unitCost Decimal @db.Decimal(18, 2)

  /**
   * Quantity originally received into this batch.
   */
  originalQuantity Decimal @db.Decimal(18, 3)

  isActive Boolean @default(true)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  company      Company       @relation(fields: [companyId], references: [id], onDelete: Cascade)
  product      Product       @relation(fields: [productId], references: [id], onDelete: Restrict)
  supplier     Supplier?     @relation(fields: [supplierId], references: [id], onDelete: SetNull)
  purchaseBill PurchaseBill? @relation(fields: [purchaseBillId], references: [id], onDelete: SetNull)

  stockBatches StockBatch[]

  purchaseLines PurchaseBillLine[]
  salesLines    SalesInvoiceLine[]

  inventoryMoves InventoryMovement[]

  @@unique([companyId, productId, batchNumber])
  @@index([companyId])
  @@index([productId])
  @@index([supplierId])
  @@index([purchaseBillId])
  @@index([batchNumber])
}

// ============================================================
// SALES INVOICE LINES
//
// IMPORTANT:
//
// The user selects the batch here.
//
// The same product may appear multiple times on one invoice
// if different batches are selected.
//
// Example:
//
// Watch ABC | Batch A | Qty 2 | Cost 1,000 | Sale 1,500
// Watch ABC | Batch B | Qty 3 | Cost 1,200 | Sale 1,600
// ============================================================

model SalesInvoiceLine {
  id        String @id @default(cuid())
  invoiceId String
  productId String
  batchId   String

  description String?

  quantity  Decimal @db.Decimal(18, 3)
  unitPrice Decimal @db.Decimal(18, 2)

  discount Decimal @default(0) @db.Decimal(18, 2)
  tax      Decimal @default(0) @db.Decimal(18, 2)
  total    Decimal @default(0) @db.Decimal(18, 2)

  warehouseId String

  /**
   * Snapshot of the batch cost at the time of sale.
   * We keep this even though InventoryBatch.unitCost exists.
   * This protects historical accounting if a batch is ever
   * corrected or adjusted later.
   */
  cogsUnitCost Decimal @default(0) @db.Decimal(18, 2)
  cogsTotal    Decimal @default(0) @db.Decimal(18, 2)

  invoice   SalesInvoice   @relation(fields: [invoiceId], references: [id], onDelete: Cascade)
  product   Product        @relation(fields: [productId], references: [id], onDelete: Restrict)
  batch     InventoryBatch @relation(fields: [batchId], references: [id], onDelete: Restrict)
  warehouse Warehouse      @relation(fields: [warehouseId], references: [id], onDelete: Restrict)

  @@index([invoiceId])
  @@index([productId])
  @@index([batchId])
  @@index([warehouseId])
}
