model StorefrontQuiz {
  id               String       @id @default(cuid())
  companyId        String
  title            String       // e.g., "Apology Finder" or "Welcome Journey"
  description      String?
  isActive         Boolean      @default(true)
  isWelcomePopup   Boolean      @default(false) // If true, triggers automatically for new visitors
  createdAt        DateTime     @default(now())
  updatedAt        DateTime     @updatedAt
  
  company          Company      @relation(fields: [companyId], references: [id], onDelete: Cascade)
  nodes            QuizNode[]
  
  @@index([companyId])
}

model QuizNode {
  id               String       @id @default(cuid())
  quizId           String
  questionText     String       // e.g., "Whom are you apologizing to?"
  subtitle         String?
  isStartingNode   Boolean      @default(false) // Marks the first question in the tree
  createdAt        DateTime     @default(now())
  updatedAt        DateTime     @updatedAt
  
  quiz             StorefrontQuiz @relation(fields: [quizId], references: [id], onDelete: Cascade)
  
  // The choices available on this specific question
  choices          QuizChoice[]   @relation("NodeToChoices")
  
  // A reverse-relation: which choices from OTHER questions lead to this one?
  targetedBy       QuizChoice[]   @relation("ChoiceToNextNode")
  
  @@index([quizId])
}

model QuizChoice {
  id               String       @id @default(cuid())
  nodeId           String
  text             String       // Button text, e.g., "My Wife" or "Explore"
  
  // ROUTING LOGIC (The magic happens here)
  nextNodeId       String?      // If they click this, go to this specific question next
  resultProductId  String?      // OR, if they click this, end the quiz and show this exact product
  resultUrl        String?      // OR, end the quiz and redirect them to a specific category page
  
  createdAt        DateTime     @default(now())
  updatedAt        DateTime     @updatedAt
  
  node             QuizNode     @relation("NodeToChoices", fields: [nodeId], references: [id], onDelete: Cascade)
  nextNode         QuizNode?    @relation("ChoiceToNextNode", fields: [nextNodeId], references: [id])
  resultProduct    Product?     @relation(fields: [resultProductId], references: [id])
  
  @@index([nodeId])
  @@index([nextNodeId])
  @@index([resultProductId])
}
